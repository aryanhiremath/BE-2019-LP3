def fib_recursive(n):
    if n <= 1:
        return n
    return fib_recursive(n-1) + fib_recursive(n-2)

n = int(input("Enter the Number : "))

print("Non Recursive Fibonacci : ")
a, b = 0,1
for i in range(n):
    print(a,end=" ")
    a, b = b, a+b

print("\n \n Recursive Fibonacci : ")
for i in range(n):
    print(fib_recursive(i),end=" ")
